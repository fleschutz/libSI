16 inline bool from_string(const std::string& str, length& result)
17 {
18 double value;
19 char unit[1024];
20 if (std::sscanf(str.c_str(), "%lf%s", &value, unit) != 2)
21 return false; // not recognized
22
23 if (unit == "Gm")
24 result = 1_Gm * value;
25 else if (unit == "Mm")
26 result = 1_Mm * value;
27 else if (unit == "km")
28 result = 1_km * value;
29 else if (unit == "m")
30 result = 1_m * value;
31 else if (unit == "dm")
32 result = 1_dm * value;
33 else if (unit == "cm")
34 result = 1_cm * value;
35 else if (unit == "mm")
36 result = 1_mm * value;
37 else if (unit == "um")
38 result = 1_um * value;
39 else if (unit == "nm")
40 result = 1_nm * value;
41 else if (unit == "pm")
42 result = 1_pm * value;
43 else
44 return false; // unknown unit
45
46 return true;
47 }

109 // internal function to join and format both quantity and unit into a string.
110 inline std::string _join(long double quantity, const char* unit, const char* formatString)
111 {
112 char buf[256];
113 std::snprintf(buf, sizeof(buf), formatString, quantity, unit);
114 return std::string(buf);
