180 FUNC length distance_on_sphere_formula(angle lat1, angle long1, angle lat2, angle long2,
181 length radius = 6371.0088_km) // mean Earth radius, recommended by NOAA
182 {
183 angle dLat = lat1 - lat2; // difference in latitude
184 angle dLong = long1 - long2; // difference in longitude
185
186 angle a = sin2(dLat / 2.) + cos(lat1) * cos(lat2) * sin2(dLong / 2.); // Haversine intermediate
187 auto c = 2. * atan2(sqrt(a), sqrt(1. - a)); // central angle
188
189 return c * radius;
190 }
